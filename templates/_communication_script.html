function communicationComposer(form) { const purpose = form.querySelector("[data-message-purpose]"); const target = form.querySelector("[data-reply-target]"); const targetField = form.querySelector("[data-reply-target-field]"); const help = form.querySelector("[data-purpose-help]"); const messageId = form.querySelector("[data-client-message-id]"); const text = form.querySelector("textarea"); let draftKey = ""; let retryAfterFailure = false; let dispatchBlocked = false; function freshId() { return window.crypto && typeof window.crypto.randomUUID === "function" ? window.crypto.randomUUID() : ""; } function refreshPurpose() { const answering = purpose.value === "workshop_answer"; targetField.hidden = !answering; target.disabled = !answering; target.required = answering; help.textContent = { workshop_answer: "Eine Antwort erledigt nur die ausgewählte Kundenfrage.", workshop_question: "Der Assistent pausiert. Die nächste Kundenantwort wird dieser Frage zugeordnet.", workshop_notification: "Eine Information ändert weder offene Kundenfragen noch den Ticketstatus." }[purpose.value] || "Bitte wählen Sie den Zweck der Nachricht."; } function saveDraft() { if (!draftKey) return; try { window.sessionStorage.setItem(draftKey, JSON.stringify({ purpose: purpose.value, reply_to_message_id: target.value, message_id: messageId.value, text: text.value })); } catch (_) {} } function changed() { if (!dispatchBlocked) messageId.value = freshId(); refreshPurpose(); saveDraft(); } purpose.addEventListener("change", changed); target.addEventListener("change", changed); text.addEventListener("input", changed); form.addEventListener("submit", (event) => { if (event.defaultPrevented) return; if (dispatchBlocked) { event.preventDefault(); return; } const alternativeAction = event.submitter && event.submitter.getAttribute("formaction"); if (alternativeAction) return; if (retryAfterFailure) { messageId.value = freshId(); retryAfterFailure = false; saveDraft(); } }); return { configure(questions, key, clearDraft, options = {}) { draftKey = key + ":communication"; retryAfterFailure = options.resultStatus === "failed"; dispatchBlocked = Boolean(options.dispatchBlocked) || ["unknown", "pending", "sending"].includes(options.resultStatus); const answerOption = purpose.querySelector("option[value='workshop_answer']"); answerOption.disabled = questions.length === 0; target.replaceChildren(new Option("Kundenfrage auswählen …", "")); for (const question of questions) { const date = question.created_at_display || question.created_at || ""; target.add(new Option(`${date}${date ? " · " : ""}${question.text || "Kundenfrage"}`, question.message_id)); } purpose.value = questions.length ? "workshop_answer" : "workshop_notification"; target.value = questions.length === 1 ? questions[0].message_id : ""; messageId.value = freshId(); try { if (clearDraft) window.sessionStorage.removeItem(draftKey); const saved = JSON.parse(window.sessionStorage.getItem(draftKey) || "null"); if (saved && saved.text === text.value) { const validPurpose = Array.from(purpose.options).some(option => option.value === saved.purpose && !option.disabled); const validTarget = saved.purpose !== "workshop_answer" || !saved.reply_to_message_id || questions.some(question => question.message_id === saved.reply_to_message_id); if (validPurpose && validTarget) { purpose.value = saved.purpose; target.value = saved.reply_to_message_id || ""; messageId.value = saved.message_id || freshId(); } else { purpose.value = ""; target.value = ""; } } } catch (_) {} refreshPurpose(); saveDraft(); } }; } function installDispatchRecovery(root) { root.addEventListener("submit", (event) => { const form = event.target; if (!form.matches("form[data-dispatch-resolution]") || event.defaultPrevented) return; const confirmation = event.submitter && event.submitter.dataset.resolutionConfirm; if (!confirmation || !window.confirm(confirmation)) event.preventDefault(); }); } function simpleDispatchAttempt(form) { const messageId = form.querySelector("input[name='message_id']"); const phone = form.querySelector("input[name='customer_phone']"); const text = form.querySelector("textarea"); let draftKey = ""; let retryAfterFailure = false; let uncertainResult = false; let action = ""; function freshId() { return window.crypto && typeof window.crypto.randomUUID === "function" ? window.crypto.randomUUID() : ""; } function blocked() { return uncertainResult || form.dataset.dispatchBlocked === "true"; } function save() { if (!draftKey) return; try { window.sessionStorage.setItem(draftKey, JSON.stringify({ message_id: messageId.value, phone: phone ? phone.value : "", text: text ? text.value : "", action })); } catch (_) {} } function changed() { if (!blocked()) messageId.value = freshId(); save(); } if (phone) phone.addEventListener("input", changed); if (text) text.addEventListener("input", changed); form.addEventListener("submit", (event) => { if (event.defaultPrevented) return; if (blocked()) { event.preventDefault(); return; } const nextAction = (event.submitter && event.submitter.getAttribute("formaction")) || form.getAttribute("action") || ""; if (retryAfterFailure || (action && nextAction !== action)) messageId.value = freshId(); retryAfterFailure = false; action = nextAction; save(); }); return { configure(key, options = {}) { draftKey = key + ":attempt"; retryAfterFailure = options.resultStatus === "failed"; uncertainResult = ["unknown", "pending", "sending"].includes(options.resultStatus); messageId.value = freshId(); action = ""; try { if (["sent", "template_sent", "success"].includes(options.resultStatus)) window.sessionStorage.removeItem(draftKey); const saved = JSON.parse(window.sessionStorage.getItem(draftKey) || "null"); if (saved) { if (options.restoreFields) { if (phone) phone.value = saved.phone || ""; if (text) text.value = saved.text || ""; } if ((!phone || saved.phone === phone.value) && (!text || saved.text === text.value)) { messageId.value = saved.message_id || freshId(); action = saved.action || ""; } } } catch (_) {} save(); } }; }
