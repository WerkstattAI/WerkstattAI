function pruefeMetaEreignis(event) { if (!['https://www.facebook.com', 'https://web.facebook.com'].includes(event.origin)) return null; let inhalt; try { inhalt = typeof event.data === 'string' ? JSON.parse(event.data) : event.data; } catch { return null; } if (!inhalt || typeof inhalt !== 'object' || Array.isArray(inhalt) || inhalt.type !== 'WA_EMBEDDED_SIGNUP' || inhalt.version !== 3) return null; const daten = inhalt.data; if (!daten || typeof daten !== 'object' || Array.isArray(daten)) return null; if (inhalt.event === 'CANCEL' || inhalt.event === 'ERROR') return {abbruch:true}; if (inhalt.event !== 'FINISH_WHATSAPP_BUSINESS_APP_ONBOARDING') return {abbruch:true}; const id = wert => typeof wert === 'string' && /^[0-9]{1,32}$/.test(wert); if (!id(daten.waba_id) || (daten.phone_number_id !== undefined && !id(daten.phone_number_id))) return null; return {meta_ursprung:event.origin, ereignis:inhalt.event, waba_id:daten.waba_id, phone_number_id:daten.phone_number_id ?? null}; } function starteAnschlussSeite(konfiguration) { const el = id => document.getElementById(id); const speicher = 'werkstattai-anschluss-' + konfiguration.workshopId; const neuerLauf = kennung => ({kennung, aktiv:false, code:null, auswahl:null, timer:null, vorbereitet:false, anzeige:0}); let gespeicherteKennung = null; try { gespeicherteKennung = sessionStorage.getItem(speicher); } catch {} let aktuell = neuerLauf(gespeicherteKennung), sdkBereit = false; const gilt = lauf => aktuell === lauf; const wechseln = kennung => { clearTimeout(aktuell.timer); aktuell = neuerLauf(kennung); return aktuell; }; const anzeigen = (lauf, text) => { if (!gilt(lauf)) return; lauf.anzeige++; el('fortschritt').textContent = text; }; // Jede Anfrage verwendet ausschließlich die Kennung ihres unveränderlichen Laufbezugs. const anfrage = async (lauf, aktion, daten={}) => { const antwort = await fetch('/dashboard/whatsapp/verbinden/' + aktion, { method:'POST', credentials:'same-origin', headers:{'Content-Type':'application/json'}, body:JSON.stringify({workshop_id:konfiguration.workshopId, kennung:lauf.kennung, ...daten}) }); if (antwort.redirected) throw new Error('Die Sitzung ist abgelaufen. Bitte erneut anmelden.'); let ergebnis; try { ergebnis = await antwort.json(); } catch { throw new Error('Anschluss nicht bestätigt. Bitte Status prüfen.'); } if (!antwort.ok) throw new Error(ergebnis.detail || 'Anschluss nicht bestätigt. Bitte Status prüfen.'); return ergebnis; }; const ergebnisAnzeigen = (lauf, ergebnis) => { if (!gilt(lauf)) return; anzeigen(lauf, ergebnis.meldung + (ergebnis.diagnose ? ' ' + ergebnis.diagnose : '')); el('fortsetzen').hidden = !ergebnis.fortsetzbar; }; const beenden = lauf => { if (!gilt(lauf)) return; lauf.aktiv=false; lauf.code=null; lauf.auswahl=null; clearTimeout(lauf.timer); el('abbruch').hidden=true; el('start').disabled=false; el('start').textContent='Neuen Anschluss vorbereiten'; }; const abbrechen = async (lauf, hinweis=null) => { if (!gilt(lauf)) return; // Auch Antworten desselben, gerade abgebrochenen Laufs dürfen nichts mehr ändern. const abbruch = wechseln(lauf.kennung); beenden(abbruch); el('fortsetzen').hidden=true; anzeigen(abbruch, 'Anschluss wird abgebrochen …'); try { if (abbruch.kennung) ergebnisAnzeigen(abbruch, await anfrage(abbruch, 'abbruch')); } catch { anzeigen(abbruch, 'Abbruch nicht bestätigt. Bitte Status prüfen; der Versuch läuft nach zehn Minuten ab.'); } if (hinweis) anzeigen(abbruch, hinweis); }; const abschliessen = async lauf => { if (!gilt(lauf) || !lauf.aktiv || !lauf.code || !lauf.auswahl) return; const daten = {code:lauf.code, ...lauf.auswahl}; lauf.aktiv=false; lauf.code=null; lauf.auswahl=null; clearTimeout(lauf.timer); anzeigen(lauf, 'Kundenzugang und Kontoabonnement werden serverseitig geprüft …'); try { ergebnisAnzeigen(lauf, await anfrage(lauf, 'abschluss', daten)); } catch (fehler) { anzeigen(lauf, fehler.message); } finally { beenden(lauf); } }; window.addEventListener('message', event => { const lauf = aktuell; if (!lauf.aktiv) return; const daten = pruefeMetaEreignis(event); if (!daten) return; if (daten.abbruch) { abbrechen(lauf); return; } lauf.auswahl=daten; abschliessen(lauf); }); // Der Versuch wird vorab erstellt; FB.login bleibt unmittelbar im Nutzerklick. const vorbereiten = async () => { const lauf = wechseln(null); el('start').disabled=true; el('fortsetzen').hidden=true; el('abbruch').hidden=true; if (window.location.origin !== konfiguration.origin) { anzeigen(lauf, 'Diese Adresse ist nicht für den Anschluss freigegeben.'); return; } anzeigen(lauf, 'Anschlussversuch wird vorbereitet …'); try { const ergebnis = await anfrage(lauf, 'start'); if (!gilt(lauf)) return; // Mit der Serverkennung beginnt ein neuer, fest gebundener Lauf. const bereit = wechseln(ergebnis.kennung); try { sessionStorage.setItem(speicher, bereit.kennung); } catch {} bereit.vorbereitet=true; el('start').disabled=false; el('start').textContent='WhatsApp verbinden'; anzeigen(bereit, 'Bereit. Mit „WhatsApp verbinden“ öffnen Sie die Meta-Anmeldung.'); } catch (fehler) { beenden(lauf); anzeigen(lauf, fehler.message); } }; el('start').addEventListener('click', () => { if (el('start').disabled || !sdkBereit || !window.FB) return; if (!aktuell.vorbereitet) { vorbereiten(); return; } const lauf = wechseln(aktuell.kennung); lauf.aktiv=true; el('start').disabled=true; el('abbruch').hidden=false; el('fortsetzen').hidden=true; anzeigen(lauf, 'Schließen Sie die Anmeldung im Meta-Fenster ab. Bei blockiertem Popup erlauben Sie Popups für diese Seite und starten erneut.'); lauf.timer=setTimeout(() => abbrechen(lauf), 9 * 60 * 1000); try { FB.login(antwort => { if (!gilt(lauf) || !lauf.aktiv) return; const wert=antwort && antwort.authResponse && antwort.authResponse.code; if (typeof wert !== 'string' || !wert || wert.length > 4096) { abbrechen(lauf); return; } lauf.code=wert; clearTimeout(lauf.timer); lauf.timer=setTimeout(() => abbrechen(lauf), 20000); abschliessen(lauf); }, {config_id:konfiguration.configId, response_type:'code', override_default_response_type:true, extras:{setup:{}, featureType:'whatsapp_business_app_onboarding', sessionInfoVersion:'3'}}); } catch { abbrechen(lauf, 'Meta-Anmeldung konnte nicht geöffnet werden. Bitte die Seite neu laden und Popups erlauben.'); } }); el('abbruch').addEventListener('click', () => abbrechen(aktuell)); const statusPruefen = async (lauf, initial=false) => { const anzeige = ++lauf.anzeige; try { if (!lauf.kennung) return; const ergebnis = await anfrage(lauf, 'status'); if (!gilt(lauf) || lauf.anzeige !== anzeige) return; ergebnisAnzeigen(lauf, ergebnis); if (initial) beenden(lauf); } catch (fehler) { if (!gilt(lauf) || lauf.anzeige !== anzeige) return; if (initial) vorbereiten(); else anzeigen(lauf, fehler.message); } }; el('status').addEventListener('click', () => statusPruefen(aktuell)); el('fortsetzen').addEventListener('click', async () => { if (el('fortsetzen').hidden) return; const lauf = wechseln(aktuell.kennung); el('fortsetzen').hidden=true; el('start').disabled=true; el('abbruch').hidden=false; anzeigen(lauf, 'Kontoabonnement wird erneut geprüft …'); try { ergebnisAnzeigen(lauf, await anfrage(lauf, 'fortsetzen')); } catch (fehler) { anzeigen(lauf, fehler.message); } finally { beenden(lauf); } }); const laden = aktuell; const ladeTimer=setTimeout(() => { if (!sdkBereit) anzeigen(laden, 'Meta-Anmeldung lädt nicht. Bitte Verbindung und Browser-Schutz prüfen.'); }, 15000); window.fbAsyncInit = () => { if (sdkBereit) return; clearTimeout(ladeTimer); try { FB.init({appId:konfiguration.appId, autoLogAppEvents:false, xfbml:false, version:konfiguration.version}); sdkBereit=true; if (!gilt(laden)) return; if (aktuell.kennung) { beenden(aktuell); statusPruefen(aktuell, true); } else vorbereiten(); } catch { anzeigen(laden, 'Meta-SDK konnte nicht initialisiert werden. Bitte später erneut versuchen.'); } }; const sdk=document.createElement('script'); sdk.src='https://connect.facebook.net/de_DE/sdk.js'; sdk.async=true; sdk.onerror=() => { clearTimeout(ladeTimer); if (!sdkBereit) anzeigen(laden, 'Meta-Anmeldung konnte nicht geladen werden. Bitte Verbindung und Browser-Schutz prüfen.'); }; document.head.appendChild(sdk); }
